    58→function WaveformBars({ isActive }: { isActive: boolean }) {
    59→  const bars = 24
    60→  return (
    61→    <div className="flex items-center justify-center gap-[3px] h-12">
    62→      {Array.from({ length: bars }).map((_, i) => (
    63→        <motion.div
    64→          key={i}
    65→          className="w-[3px] rounded-full bg-gradient-to-t from-primary/60 to-primary"
    66→          animate={
    67→            isActive
    68→              ? {
    69→                  height: [8, Math.random() * 40 + 8, 8],
    70→                }
    71→              : { height: 8 }
    72→          }
    73→          transition={
    74→            isActive
    75→              ? {
    76→                  duration: 0.6 + Math.random() * 0.4,
    77→                  repeat: Infinity,
    78→                  repeatType: 'reverse',
    79→                  delay: i * 0.03,
    80→                  ease: 'easeInOut',
    81→                }
    82→              : { duration: 0.3 }
    83→          }
    84→        />
    85→      ))}
    86→    </div>
    87→  )
    88→}

   145→
   146→export function AddMemorySheet({ open, onOpenChange }: AddMemorySheetProps) {
   147→  const isMobile = useIsMobile()
   148→  const memories = useAetherStore((s) => s.memories)
   149→  const isAuthenticated = useAetherStore((s) => s.isAuthenticated)
   150→  const setShowAuthModal = useAetherStore((s) => s.setShowAuthModal)
   151→  const requireAuth = useAetherStore((s) => s.requireAuth)
   152→
   153→  const {
   154→    captureText,
   155→    captureLink,
   156→    captureImage,
   157→    captureVoice,
   158→    isRecording,
   159→    startRecording,
   160→    stopRecording,
   161→    recordingDuration,
   162→    isCapturing,
   163→  } = useCapture()
   164→
   165→  // ── Free plan limit ────────────────────────────────────────────────
   166→  const FREE_MEMORY_LIMIT = 15
   167→
   168→  // Text tab state
   169→  const [title, setTitle] = useState('')
   170→  const [content, setContent] = useState('')
   171→
   172→  // Link tab state
   173→  const [url, setUrl] = useState('')
   174→  const [linkTitle, setLinkTitle] = useState('')
   175→  const [linkNotes, setLinkNotes] = useState('')
   176→  const [isFetchingTitle, setIsFetchingTitle] = useState(false)
   177→
   178→  // Image tab state
   179→  const [imageFile, setImageFile] = useState<File | null>(null)
   180→  const [imagePreviewUrl, setImagePreviewUrl] = useState<string | null>(null)
   181→  const [imageTitle, setImageTitle] = useState('')
   182→  const [isDragging, setIsDragging] = useState(false)
   183→  const fileInputRef = useRef<HTMLInputElement>(null)
   184→
   185→  // Voice tab state
   186→  const [voiceTitle, setVoiceTitle] = useState('')
   187→  const [recordedBlob, setRecordedBlob] = useState<Blob | null>(null)
   188→  const [isPlaying, setIsPlaying] = useState(false)
   189→  const audioRef = useRef<HTMLAudioElement | null>(null)
   190→
   191→  // Common state
   192→  const [activeTab, setActiveTab] = useState('text')
   193→  const [showSuccess, setShowSuccess] = useState(false)
   194→
   195→  // ── Cleanup image preview URL ──────────────────────────────────────
   196→  useEffect(() => {
   197→    return () => {
   198→      if (imagePreviewUrl) URL.revokeObjectURL(imagePreviewUrl)
   199→    }
   200→  }, [imagePreviewUrl])
   201→
   202→  // ── Format recording duration ──────────────────────────────────────
   203→  const formatDuration = (seconds: number) => {
   204→    const m = Math.floor(seconds / 60)
   205→    const s = seconds % 60
   206→    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   207→  }
   208→
   209→  // ── Reset form ─────────────────────────────────────────────────────
   210→  const resetForm = useCallback(() => {
   211→    setTitle('')
   212→    setContent('')
   213→    setUrl('')
   214→    setLinkTitle('')
   215→    setLinkNotes('')
   216→    setImageFile(null)
   217→    if (imagePreviewUrl) URL.revokeObjectURL(imagePreviewUrl)
   218→    setImagePreviewUrl(null)
   219→    setImageTitle('')
   220→    setVoiceTitle('')
   221→    setRecordedBlob(null)
   222→    setIsPlaying(false)
   223→    setActiveTab('text')
   224→    setShowSuccess(false)
   225→    if (audioRef.current) {
   226→      audioRef.current.pause()
   227→      audioRef.current = null
   228→    }
   229→  }, [imagePreviewUrl])
   230→
   231→  const handleClose = useCallback(() => {
   232→    onOpenChange(false)
   233→    setTimeout(resetForm, 300)
   234→  }, [onOpenChange, resetForm])
   235→
   236→  // ── Fetch link title ───────────────────────────────────────────────
   237→  const fetchLinkTitle = useCallback(async () => {
   238→    if (!url.trim()) return
   239→    setIsFetchingTitle(true)
   240→    try {
   241→      const res = await fetch('/api/memories/fetch-title', {
   242→        method: 'POST',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({ url: url.trim() }),
   245→      })
   246→      if (res.ok) {
   247→        const data = await res.json()
   248→        if (data.title) {
   249→          setLinkTitle(data.title)
   250→        }
   251→      }
   252→    } catch {
   253→      // Silently fail
   254→    } finally {
   255→      setIsFetchingTitle(false)
   256→    }
   257→  }, [url])
   258→
   259→  // ── Image file selection ───────────────────────────────────────────
   260→  const handleImageSelect = useCallback((file: File) => {
   261→    if (!file.type.startsWith('image/')) {
   262→      toast.error('Only image files are accepted')
   263→      return
   264→    }
   265→    if (file.size > 10 * 1024 * 1024) {
   266→      toast.error('Image must be under 10MB')
   267→      return
   268→    }
   269→    if (imagePreviewUrl) URL.revokeObjectURL(imagePreviewUrl)
   270→    const url = URL.createObjectURL(file)
   271→    setImageFile(file)
   272→    setImagePreviewUrl(url)
   273→  }, [imagePreviewUrl])
   274→
   275→  const handleFileInput = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   276→    const file = e.target.files?.[0]
   277→    if (file) handleImageSelect(file)
   278→  }, [handleImageSelect])
   279→
   280→  const removeImage = useCallback(() => {
   281→    if (imagePreviewUrl) URL.revokeObjectURL(imagePreviewUrl)
   282→    setImageFile(null)
   283→    setImagePreviewUrl(null)
   284→    if (fileInputRef.current) fileInputRef.current.value = ''
   285→  }, [imagePreviewUrl])
   286→
   287→  // ── Drag and drop handlers ─────────────────────────────────────────
   288→  const handleDragOver = useCallback((e: React.DragEvent) => {
   289→    e.preventDefault()
   290→    setIsDragging(true)
   291→  }, [])
   292→
   293→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   294→    e.preventDefault()
   295→    setIsDragging(false)
   296→  }, [])
   297→
   298→  const handleDrop = useCallback((e: React.DragEvent) => {
   299→    e.preventDefault()
   300→    setIsDragging(false)
   301→    const file = e.dataTransfer.files[0]
   302→    if (file) handleImageSelect(file)
   303→  }, [handleImageSelect])
   304→
   305→  // ── Voice recording ────────────────────────────────────────────────
   306→  const handleStartRecording = useCallback(async () => {
   307→    if (!isAuthenticated) {
   308→      setShowAuthModal(true)
   309→      toast.error('Sign in to capture memories')
   310→      return
   311→    }
   312→    try {
   313→      setRecordedBlob(null)
   314→      setIsPlaying(false)
   315→      if (audioRef.current) {
   316→        audioRef.current.pause()
   317→        audioRef.current = null
   318→      }
   319→      await startRecording()
   320→    } catch {
   321→      toast.error('Could not start recording')
   322→    }
   323→  }, [isAuthenticated, setShowAuthModal, startRecording])
   324→
   325→  const handleStopRecording = useCallback(async () => {
   326→    const blob = await stopRecording()
   327→    if (blob) {
   328→      setRecordedBlob(blob)
   329→    }
   330→  }, [stopRecording])
   331→
   332→  // ── Voice playback ─────────────────────────────────────────────────
   333→  const togglePlayback = useCallback(() => {
   334→    if (!recordedBlob) return
   335→
   336→    if (isPlaying && audioRef.current) {
   337→      audioRef.current.pause()
   338→      audioRef.current.currentTime = 0
   339→      setIsPlaying(false)
   340→      return
   341→    }
   342→
   343→    const audioUrl = URL.createObjectURL(recordedBlob)
   344→    const audio = new Audio(audioUrl)
   345→    audio.onended = () => {
   346→      setIsPlaying(false)
   347→      URL.revokeObjectURL(audioUrl)
   348→    }
   349→    audio.play()
   350→    audioRef.current = audio
   351→    setIsPlaying(true)
   352→  }, [recordedBlob, isPlaying])
   353→
   354→  // ── Save handlers ──────────────────────────────────────────────────
   355→  const showSuccessAnimation = useCallback(() => {
   356→    setShowSuccess(true)
   357→    setTimeout(() => setShowSuccess(false), 2000)
   358→  }, [])
   359→
   360→  const handleSave = useCallback(async () => {
   361→    // Free plan paywall
   362→    if (memories.length >= FREE_MEMORY_LIMIT) {
   363→      toast.error('Free limit reached — upgrade to Pro for unlimited memories', {
   364→        icon: <Crown className="size-4" />,
   365→      })
   366→      return
   367→    }
   368→
   369→    // Auth gate
   370→    if (!isAuthenticated) {
   371→      requireAuth(() => {
   372→        // After auth, the user can try saving again
   373→        setShowAuthModal(true)
   374→      })
   375→      handleClose()
   376→      return
   377→    }
   378→
   379→    if (activeTab === 'text') {
   380→      if (!content.trim()) return
   381→      const result = await captureText(title.trim() || 'Untitled Note', content.trim())
   382→      if (result.success) {
   383→        showSuccessAnimation()
   384→        setTimeout(handleClose, 1200)
   385→        toast.success('Memory captured')
   386→      } else {
   387→        toast.error(result.error || 'Failed to save')
   388→      }
   389→    } else if (activeTab === 'link') {
   390→      if (!url.trim()) return
   391→      const result = await captureLink(url.trim(), linkTitle.trim() || undefined, linkNotes.trim() || undefined)
   392→      if (result.success) {
   393→        showSuccessAnimation()
   394→        setTimeout(handleClose, 1200)
   395→        toast.success('Link captured')
   396→      } else {
   397→        toast.error(result.error || 'Failed to save')
   398→      }
   399→    } else if (activeTab === 'image') {
   400→      if (!imageFile) {
   401→        toast.error('Please select an image')
   402→        return
   403→      }
   404→      const result = await captureImage(imageFile, imageTitle.trim() || undefined)
   405→      if (result.success) {
   406→        showSuccessAnimation()
   407→        setTimeout(handleClose, 1200)
   408→        toast.success('Image captured')
   409→      } else {
   410→        toast.error(result.error || 'Failed to save')
   411→      }
   412→    } else if (activeTab === 'voice') {
   413→      if (!recordedBlob) {
   414→        toast.error('Please record audio first')
   415→        return
   416→      }
   417→      const result = await captureVoice(recordedBlob, voiceTitle.trim() || undefined)
   418→      if (result.success) {
   419→        showSuccessAnimation()
   420→        setTimeout(handleClose, 1200)
   421→        toast.success('Voice note captured')
   422→      } else {
   423→        toast.error(result.error || 'Failed to save')
   424→      }
   425→    }
   426→  }, [
   427→    activeTab, title, content, url, linkTitle, linkNotes,
   428→    imageFile, imageTitle, recordedBlob, voiceTitle,
   429→    isAuthenticated, memories.length, requireAuth, handleClose,
   430→    captureText, captureLink, captureImage, captureVoice,
   431→    showSuccessAnimation,
   432→  ])
   433→
   434→  // ── Determine if save is disabled ──────────────────────────────────
   435→  const isSaveDisabled = () => {
   436→    if (isCapturing || isRecording) return true
   437→    if (activeTab === 'text') return !content.trim()
   438→    if (activeTab === 'link') return !url.trim()
   439→    if (activeTab === 'image') return !imageFile
   440→    if (activeTab === 'voice') return !recordedBlob
   441→    return false
   442→  }
   443→
   444→  // ═════════════════════════════════════════════════════════════════
   445→  // ─── TAB CONTENT ──────────────────────────────────────────────────
   446→  // ═════════════════════════════════════════════════════════════════
   447→
   448→  const tabContent = (
   449→    <div className="relative">
   450→      <SuccessOverlay show={showSuccess} />
   451→
   452→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   453→        <TabsList className="w-full grid grid-cols-4 bg-white/50 dark:bg-white/5 backdrop-blur-sm border border-white/20">
   454→          <TabsTrigger value="text" className="gap-1.5 data-[state=active]:bg-primary/10 data-[state=active]:text-primary transition-all">
   455→            <FileText className="size-4" />
   456→            <span className="hidden sm:inline">Text</span>
   457→          </TabsTrigger>
   458→          <TabsTrigger value="link" className="gap-1.5 data-[state=active]:bg-primary/10 data-[state=active]:text-primary transition-all">
   459→            <Link2 className="size-4" />
   460→            <span className="hidden sm:inline">Link</span>
   461→          </TabsTrigger>
   462→          <TabsTrigger value="image" className="gap-1.5 data-[state=active]:bg-primary/10 data-[state=active]:text-primary transition-all">
   463→            <ImageIcon className="size-4" />
   464→            <span className="hidden sm:inline">Image</span>
   465→          </TabsTrigger>
   466→          <TabsTrigger value="voice" className="gap-1.5 data-[state=active]:bg-primary/10 data-[state=active]:text-primary transition-all">
   467→            <Mic className="size-4" />
   468→            <span className="hidden sm:inline">Voice</span>
   469→          </TabsTrigger>
   470→        </TabsList>
   471→
   472→        {/* ── Text Tab ─────────────────────────────────────────────── */}
   473→        <TabsContent value="text" className="space-y-4 pt-4">
   474→          <motion.div
   475→            initial={{ opacity: 0, y: 8 }}
   476→            animate={{ opacity: 1, y: 0 }}
   477→            transition={{ duration: 0.25 }}
   478→            className="space-y-4"
   479→          >
   480→            <div className="space-y-2">
   481→              <Label htmlFor="memory-title" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Title</Label>
   482→              <Input
   483→                id="memory-title"
   484→                placeholder="Give your memory a title..."
   485→                value={title}
   486→                onChange={(e) => setTitle(e.target.value)}
   487→                className="bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   488→              />
   489→            </div>
   490→            <div className="space-y-2">
   491→              <Label htmlFor="memory-content" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Content</Label>
   492→              <Textarea
   493→                id="memory-content"
   494→                placeholder="Write your thoughts, notes, ideas..."
   495→                value={content}
   496→                onChange={(e) => setContent(e.target.value)}
   497→                className="min-h-[160px] resize-y bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   498→              />
   499→            </div>
   500→          </motion.div>
   501→        </TabsContent>
   502→
   503→        {/* ── Link Tab ─────────────────────────────────────────────── */}
   504→        <TabsContent value="link" className="space-y-4 pt-4">
   505→          <motion.div
   506→            initial={{ opacity: 0, y: 8 }}
   507→            animate={{ opacity: 1, y: 0 }}
   508→            transition={{ duration: 0.25 }}
   509→            className="space-y-4"
   510→          >
   511→            <div className="space-y-2">
   512→              <Label htmlFor="link-url" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">URL</Label>
   513→              <div className="flex gap-2">
   514→                <Input
   515→                  id="link-url"
   516→                  placeholder="https://..."
   517→                  value={url}
   518→                  onChange={(e) => setUrl(e.target.value)}
   519→                  onBlur={() => fetchLinkTitle()}
   520→                  onKeyDown={(e) => {
   521→                    if (e.key === 'Enter') fetchLinkTitle()
   522→                  }}
   523→                  className="bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   524→                />
   525→                <Button
   526→                  variant="outline"
   527→                  size="icon"
   528→                  onClick={fetchLinkTitle}
   529→                  disabled={isFetchingTitle || !url.trim()}
   530→                  className="shrink-0 bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 hover:bg-primary/10 hover:border-primary/30 transition-colors"
   531→                >
   532→                  {isFetchingTitle ? (
   533→                    <Loader2 className="size-4 animate-spin" />
   534→                  ) : (
   535→                    <Link2 className="size-4" />
   536→                  )}
   537→                </Button>
   538→              </div>
   539→            </div>
   540→            <div className="space-y-2">
   541→              <Label htmlFor="link-title" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Title</Label>
   542→              <Input
   543→                id="link-title"
   544→                placeholder="Link title (auto-fetched or type manually)"
   545→                value={linkTitle}
   546→                onChange={(e) => setLinkTitle(e.target.value)}
   547→                className="bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   548→              />
   549→            </div>
   550→            <div className="space-y-2">
   551→              <Label htmlFor="link-notes" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Notes</Label>
   552→              <Textarea
   553→                id="link-notes"
   554→                placeholder="Add notes about this link..."
   555→                value={linkNotes}
   556→                onChange={(e) => setLinkNotes(e.target.value)}
   557→                className="min-h-[100px] resize-y bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   558→              />
   559→            </div>
   560→          </motion.div>
   561→        </TabsContent>
   562→
   563→        {/* ── Image Tab ────────────────────────────────────────────── */}
   564→        <TabsContent value="image" className="space-y-4 pt-4">
   565→          <motion.div
   566→            initial={{ opacity: 0, y: 8 }}
   567→            animate={{ opacity: 1, y: 0 }}
   568→            transition={{ duration: 0.25 }}
   569→            className="space-y-4"
   570→          >
   571→            <div className="space-y-2">
   572→              <Label htmlFor="image-title" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Title</Label>
   573→              <Input
   574→                id="image-title"
   575→                placeholder="Image title..."
   576→                value={imageTitle}
   577→                onChange={(e) => setImageTitle(e.target.value)}
   578→                className="bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   579→              />
   580→            </div>
   581→
   582→            {!imagePreviewUrl ? (
   583→              <div
   584→                onDragOver={handleDragOver}
   585→                onDragLeave={handleDragLeave}
   586→                onDrop={handleDrop}
   587→                onClick={() => fileInputRef.current?.click()}
   588→                className={cn(
   589→                  'relative flex flex-col items-center justify-center rounded-xl border-2 border-dashed p-8 text-center cursor-pointer transition-all duration-300',
   590→                  'bg-white/30 dark:bg-white/5 backdrop-blur-sm',
   591→                  isDragging
   592→                    ? 'border-primary/60 bg-primary/5 scale-[1.02]'
   593→                    : 'border-white/30 hover:border-primary/40 hover:bg-primary/5',
   594→                )}
   595→              >
   596→                <input
   597→                  ref={fileInputRef}
   598→                  type="file"
   599→                  accept="image/*"
   600→                  className="hidden"
   601→                  onChange={handleFileInput}
   602→                />
   603→                <motion.div
   604→                  animate={isDragging ? { scale: 1.1, y: -4 } : { scale: 1, y: 0 }}
   605→                  transition={{ type: 'spring', stiffness: 300, damping: 25 }}
   606→                >
   607→                  <Upload className="size-10 text-muted-foreground/60 mb-3" />
   608→                </motion.div>
   609→                <p className="text-sm font-medium text-muted-foreground">
   610→                  {isDragging ? 'Drop your image here' : 'Drop an image or click to upload'}
   611→                </p>
   612→                <p className="text-xs text-muted-foreground/60 mt-1">
   613→                  PNG, JPG, GIF, WebP — up to 10MB
   614→                </p>
   615→              </div>
   616→            ) : (
   617→              <motion.div
   618→                initial={{ opacity: 0, scale: 0.95 }}
   619→                animate={{ opacity: 1, scale: 1 }}
   620→                className="relative group"
   621→              >
   622→                <div className="rounded-xl overflow-hidden border border-white/20 shadow-[0_20px_50px_rgba(109,89,122,0.08)]">
   623→                  <img
   624→                    src={imagePreviewUrl}
   625→                    alt="Preview"
   626→                    className="w-full h-auto max-h-[280px] object-cover"
   627→                  />
   628→                </div>
   629→                <Button
   630→                  variant="destructive"
   631→                  size="icon"
   632→                  className="absolute top-2 right-2 size-8 opacity-0 group-hover:opacity-100 transition-opacity bg-black/50 hover:bg-black/70 backdrop-blur-sm border-0"
   633→                  onClick={removeImage}
   634→                >
   635→                  <X className="size-4" />
   636→                </Button>
   637→              </motion.div>
   638→            )}
   639→          </motion.div>
   640→        </TabsContent>
   641→
   642→        {/* ── Voice Tab ────────────────────────────────────────────── */}
   643→        <TabsContent value="voice" className="space-y-4 pt-4">
   644→          <motion.div
   645→            initial={{ opacity: 0, y: 8 }}
   646→            animate={{ opacity: 1, y: 0 }}
   647→            transition={{ duration: 0.25 }}
   648→            className="space-y-4"
   649→          >
   650→            <div className="space-y-2">
   651→              <Label htmlFor="voice-title" className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Title</Label>
   652→              <Input
   653→                id="voice-title"
   654→                placeholder="Voice note title..."
   655→                value={voiceTitle}
   656→                onChange={(e) => setVoiceTitle(e.target.value)}
   657→                className="bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 focus:border-primary/40 transition-colors"
   658→              />
   659→            </div>
   660→
   661→            <div className="rounded-xl border border-white/20 bg-white/30 dark:bg-white/5 backdrop-blur-xl p-6 shadow-[0_20px_50px_rgba(109,89,122,0.05)]">
   662→              {/* Waveform visualization */}
   663→              <WaveformBars isActive={isRecording} />
   664→
   665→              {/* Timer */}
   666→              <div className="text-center mt-3">
   667→                <motion.span
   668→                  className={cn(
   669→                    'text-2xl font-mono font-light tracking-widest',
   670→                    isRecording ? 'text-red-500' : 'text-muted-foreground'
   671→                  )}
   672→                  animate={isRecording ? { opacity: [1, 0.5, 1] } : { opacity: 1 }}
   673→                  transition={{ duration: 1.5, repeat: Infinity }}
   674→                >
   675→                  {formatDuration(recordingDuration)}
   676→                </motion.span>
   677→              </div>
   678→
   679→              {/* Controls */}
   680→              <div className="flex items-center justify-center gap-4 mt-5">
   681→                {!isRecording ? (
   682→                  <motion.div whileTap={{ scale: 0.92 }}>
   683→                    <Button
   684→                      onClick={handleStartRecording}
   685→                      size="lg"
   686→                      className={cn(
   687→                        'rounded-full size-14 p-0 shadow-lg transition-all',
   688→                        'bg-gradient-to-br from-red-500 to-red-600 hover:from-red-600 hover:to-red-700 text-white',
   689→                        'shadow-red-500/20 hover:shadow-red-500/30'
   690→                      )}
   691→                    >
   692→                      <Mic className="size-6" />
   693→                    </Button>
   694→                  </motion.div>
   695→                ) : (
   696→                  <motion.div whileTap={{ scale: 0.92 }}>
   697→                    <Button
   698→                      onClick={handleStopRecording}
   699→                      size="lg"
   700→                      className="rounded-full size-14 p-0 shadow-lg bg-red-500 hover:bg-red-600 text-white shadow-red-500/20"
   701→                    >
   702→                      <Square className="size-5" />
   703→                    </Button>
   704→                  </motion.div>
   705→                )}
   706→
   707→                {/* Playback button */}
   708→                {recordedBlob && !isRecording && (
   709→                  <motion.div
   710→                    initial={{ opacity: 0, scale: 0.8 }}
   711→                    animate={{ opacity: 1, scale: 1 }}
   712→                    whileTap={{ scale: 0.92 }}
   713→                  >
   714→                    <Button
   715→                      onClick={togglePlayback}
   716→                      variant="outline"
   717→                      size="lg"
   718→                      className="rounded-full size-12 p-0 bg-white/50 dark:bg-white/5 backdrop-blur-sm border-white/20 hover:bg-primary/10 hover:border-primary/30"
   719→                    >
   720→                      {isPlaying ? (
   721→                        <Square className="size-4" />
   722→                      ) : (
   723→                        <Play className="size-4 ml-0.5" />
   724→                      )}
   725→                    </Button>
   726→                  </motion.div>
   727→                )}
   728→              </div>
   729→
   730→              {/* Status text */}
   731→              <p className="text-center text-xs text-muted-foreground mt-3">
   732→                {isRecording
   733→                  ? 'Tap to stop recording'
   734→                  : recordedBlob
   735→                    ? 'Recording ready — listen back or save'
   736→                    : 'Tap the microphone to start recording'}
   737→              </p>
   738→            </div>
   739→          </motion.div>
   740→        </TabsContent>
   741→      </Tabs>
   742→
   743→      {/* ── Save Button ────────────────────────────────────────────── */}
   744→      <div className="pt-5 pb-2">
   745→        <Button
   746→          onClick={handleSave}
   747→          disabled={isSaveDisabled()}
   748→          className={cn(
   749→            'w-full relative overflow-hidden transition-all duration-300',
   750→            'bg-gradient-to-r from-primary to-[#8B6F9A] text-primary-foreground',
   751→            'hover:opacity-90 hover:shadow-lg hover:shadow-primary/20',
   752→            isCapturing && 'cursor-wait',
   753→          )}
   754→        >
   755→          {isCapturing ? (
   756→            <motion.div
   757→              className="flex items-center gap-2"
   758→              initial={{ opacity: 0, y: 4 }}
   759→              animate={{ opacity: 1, y: 0 }}
   760→            >
   761→              <Loader2 className="size-4 animate-spin" />
   762→              Saving & Synthesizing...
   763→            </motion.div>
   764→          ) : (
   765→            <motion.span
   766→              initial={{ opacity: 0 }}
   767→              animate={{ opacity: 1 }}
   768→              className="flex items-center gap-2"
   769→            >
   770→              <Sparkles className="size-4" />
   771→              Save Memory
   772→            </motion.span>
   773→          )}
   774→        </Button>
   775→      </div>
   776→    </div>
   777→  )
   778→
   779→  // ═════════════════════════════════════════════════════════════════
   780→  // ─── RESPONSIVE: SHEET (desktop) / DRAWER (mobile) ──────────────
   781→  // ═════════════════════════════════════════════════════════════════
   782→
   783→  if (isMobile) {
   784→    return (
   785→      <Drawer open={open} onOpenChange={onOpenChange}>
   786→        <DrawerContent className="bg-background/95 backdrop-blur-xl">
   787→          <DrawerHeader className="pb-2">
   788→            <DrawerTitle className="text-lg font-semibold">New Memory</DrawerTitle>
   789→            <DrawerDescription className="text-sm text-muted-foreground">
   790→              Capture a thought, link, image, or voice note.
   791→            </DrawerDescription>
   792→          </DrawerHeader>
   793→          <div className="px-4 pb-4">
   794→            {tabContent}
   795→          </div>
   796→        </DrawerContent>
   797→      </Drawer>

   799→  }
   800→
   801→  return (

   803→      <SheetContent
   804→        side="right"
   805→        className="sm:max-w-md overflow-y-auto bg-background/95 backdrop-blur-xl border-l border-white/10"
   806→      >
   807→        <SheetHeader>
   808→          <SheetTitle className="text-lg font-semibold">New Memory</SheetTitle>
