    61→function useIsMobile() {

    65→    check()

   209→// ─── WEEKLY RECAP HELPER ─────────────────────────────────────────────
   210→// ═══════════════════════════════════════════════════════════════════════
   211→
   212→function getWeeklyRecap(memories: Memory[]): { count: number; topTags: string[]; summary: string } | null {
   213→  const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000
   214→  const weekMemories = memories.filter(m => new Date(m.createdAt).getTime() > sevenDaysAgo)
   215→  if (weekMemories.length === 0) return null
   216→
   217→  const tagCounts: Record<string, number> = {}
   218→  weekMemories.forEach(m => {
   219→    m.tags.forEach(t => {
   220→      tagCounts[t] = (tagCounts[t] || 0) + 1
   221→    })
   222→  })
   223→  const topTags = Object.entries(tagCounts)
   224→    .sort((a, b) => b[1] - a[1])
   225→    .slice(0, 3)
   226→    .map(([tag]) => tag)
   227→
   228→  // Check for recap field in most recent memory
   229→  const latestRecap = weekMemories[0]?.recap
   230→  if (latestRecap) {
   231→    return { count: weekMemories.length, topTags, summary: latestRecap.slice(0, 200) }
   232→  }
   233→
   234→  // Generate a simple summary from the week's data
   235→  const summaries = weekMemories
   236→    .map(m => m.summary || m.content || m.title)
   237→    .filter(Boolean)
   238→    .slice(0, 5)
   239→  const combinedText = summaries.join('. ')
   240→  const summaryText = combinedText.length > 150
   241→    ? combinedText.slice(0, 147) + '...'
   242→    : combinedText || `You captured ${weekMemories.length} thought${weekMemories.length > 1 ? 's' : ''} this week.`
   243→
   244→  return { count: weekMemories.length, topTags, summary: summaryText }
   245→}
