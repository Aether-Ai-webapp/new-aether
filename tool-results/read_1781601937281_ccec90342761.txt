   402→   402→              memory_id: memoryId,
   403→   403→              collection_id: matched.id,
   404→   404→            })

   406→   406→        }
   407→   407→      }
   408→   408→    }

   448→   448→          },

   457→   457→    title: memory.title,

   459→   459→    summary: memory.summary,
   460→   460→    deepInsight: memory.deepInsight || null,
   461→   461→    tags: memory.tags ? memory.tags.split(',').filter(Boolean) : [],
   462→   462→    sourceUrl: memory.sourceUrl,
   463→   463→    fileUrl: memory.fileUrl || null,
   464→   464→    imagePreview: memory.imagePreview || null,
   465→   465→    imageUrl: memory.imageUrl || null,
   466→   466→    recap: memory.recap || null,
   467→   467→    isFavorite: memory.isFavorite,
   468→   468→    createdAt: memory.createdAt.toISOString(),
   469→   469→    updatedAt: memory.updatedAt.toISOString(),
   470→   470→    collections: memory.collections.map(mc => ({
   471→   471→      id: mc.collection.id,
   472→   472→      name: mc.collection.name,
   473→   473→      color: mc.collection.color,
   474→   474→      icon: mc.collection.icon,
   475→   475→    })),
   476→   476→  }

   480→   480→// ─── FIND CONNECTED MEMORIES (for AI Brain) ─────────────────────────
   481→   481→// ═══════════════════════════════════════════════════════════════════════
   482→   482→
   483→   483→async function findConnectedMemories(
   484→   484→  newMemoryTags: string[],
   485→   485→  newMemoryContent: string,
   486→   486→  newMemoryId: string
   487→   487→): Promise<{ id: string; title: string; reason: string }[]> {
   488→   488→  try {
   489→   489→    // Get all existing memories from Prisma
   490→   490→    const allMemories = await db.memory.findMany({
   491→   491→      where: {
   492→   492→        id: { not: newMemoryId },
   493→   493→      },
   494→   494→      select: {
   495→   495→        id: true,
   496→   496→        title: true,
   497→   497→        content: true,
   498→   498→        tags: true,
   499→   499→        type: true,
   500→   500→      },
   501→   501→      take: 50,
   502→   502→      orderBy: { createdAt: 'desc' },
   503→   503→    })
   504→   504→
   505→   505→    if (allMemories.length === 0) return []
   506→   506→
   507→   507→    // ── Quick tag-based matching (instant, no LLM needed) ─────────
   508→   508→    const connected: { id: string; title: string; reason: string; score: number }[] = []
   509→   509→
   510→   510→    for (const mem of allMemories) {
   511→   511→      const memTags = mem.tags ? mem.tags.split(',').filter(Boolean) : []
   512→   512→      const overlapTags = memTags.filter(t => newMemoryTags.includes(t.toLowerCase()))
   513→   513→
   514→   514→      if (overlapTags.length > 0) {
   515→   515→        connected.push({
   516→   516→          id: mem.id,
   517→   517→          title: mem.title || 'Untitled',
   518→   518→          reason: `Shares tags: ${overlapTags.join(', ')}`,
   519→   519→          score: overlapTags.length * 2,
   520→   520→        })
   521→   521→      }
   522→   522→
   523→   523→      // Content similarity — check for common meaningful words
   524→   524→      const contentWords = mem.content.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   525→   525→      const newWords = newMemoryContent.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   526→   526→      const overlapWords = contentWords.filter(w => newWords.includes(w))
   527→   527→      if (overlapWords.length >= 2) {
   528→   528→        const existing = connected.find(c => c.id === mem.id)
   529→   529→        if (existing) {
   530→   530→          existing.score += overlapWords.length
   531→   531→        } else {
   532→   532→          connected.push({
   533→   533→            id: mem.id,
   534→   534→            title: mem.title || 'Untitled',
   535→   535→            reason: `Similar content themes`,
   536→   536→            score: overlapWords.length,
   537→   537→          })
   538→   538→        }
   539→   539→      }
   540→   540→    }
   541→   541→
   542→   542→    // Sort by score and return top 5
   543→   543→    return connected
   544→   544→      .sort((a, b) => b.score - a.score)
   545→   545→      .slice(0, 5)
   546→   546→      .map(({ id, title, reason }) => ({ id, title, reason }))
   547→   547→  } catch (err) {
   548→   548→    console.warn('Find connected memories failed:', err instanceof Error ? err.message : 'Unknown')
   549→   549→    return []
   550→   550→  }
   551→   551→}

   553→   553→// ═══════════════════════════════════════════════════════════════════════
   554→   554→// ─── MAIN POST HANDLER ──────────────────────────────────────────────
   555→   555→// ═══════════════════════════════════════════════════════════════════════
   556→   556→
   557→   557→export async function POST(req: NextRequest) {
   558→   558→  try {
   559→   559→    // ── STEP 1: Parse FormData ──────────────────────────────────────
   560→   560→    const formData = await req.formData()
   561→   561→    const text = (formData.get('text') as string) || ''
   562→   562→    const url = (formData.get('url') as string) || ''
   563→   563→    const imageFile = formData.get('image') as File | null
   564→   564→    const audioFile = formData.get('audio') as File | null
   565→   565→
   566→   566→    // Validate: at least one content source
   567→   567→    const hasText = text.trim().length > 0
   568→   568→    const hasUrl = url.trim().length > 0
   569→   569→    const hasImage = imageFile && imageFile.size > 0
   570→   570→    const hasAudio = audioFile && audioFile.size > 0
   571→   571→
   572→   572→    if (!hasText && !hasUrl && !hasImage && !hasAudio) {
   573→   573→      return NextResponse.json({ error: 'No content provided' }, { status: 400 })
   574→   574→    }
   575→   575→
   576→   576→    // ── STEP 2: Process Audio → Transcript ──────────────────────────
   577→   577→    let audioTranscript = ''
   578→   578→    let memoryType: string = 'text'
   579→   579→
   580→   580→    if (hasAudio) {
   581→   581→      memoryType = 'voice'
   582→   582→      audioTranscript = await transcribeAudio(audioFile)
   583→   583→    }
   584→   584→
   585→   585→    // ── STEP 3: Analyze Image with VLM ──────────────────────────────
   586→   586→    let imageAnalysis: ImageAnalysis | null = null
   587→   587→    let imageUrl: string | null = null
   588→   588→
   589→   589→    if (hasImage) {
   590→   590→      memoryType = 'image'
   591→   591→      // Run VLM analysis on the image (always works)
   592→   592→      imageAnalysis = await analyzeImageWithVLM(imageFile)
   593→   593→    }
   594→   594→
   595→   595→    // ── STEP 4: Assemble Raw Content ────────────────────────────────
   596→   596→    let rawContent = ''
   597→   597→    if (hasText) rawContent += text.trim()
   598→   598→    if (hasUrl) rawContent += (rawContent ? '\n\n' : '') + `URL: ${url.trim()}`
   599→   599→    if (audioTranscript) rawContent += (rawContent ? '\n\n' : '') + `Voice Transcript:\n${audioTranscript}`
   600→   600→
   601→   601→    // For images, use VLM description as content (not just "Image capture")
   602→   602→    if (hasImage) {
   603→   603→      if (imageAnalysis) {
   604→   604→        const imageContentParts: string[] = []
   605→   605→        if (imageAnalysis.description) imageContentParts.push(imageAnalysis.description)
   606→   606→        if (imageAnalysis.extracted_text) imageContentParts.push(`Text in image: ${imageAnalysis.extracted_text}`)
   607→   607→        if (imageAnalysis.objects.length > 0) imageContentParts.push(`Objects: ${imageAnalysis.objects.join(', ')}`)
   608→   608→
   609→   609→        const imageContent = imageContentParts.join('\n\n')
   610→   610→        rawContent += (rawContent ? '\n\n' : '') + imageContent
   611→   611→      } else if (!rawContent) {
   612→   612→        rawContent = 'Image capture'
   613→   613→      }
   614→   614→    }
   615→   615→
   616→   616→    if (!rawContent) rawContent = 'Captured content'
   617→   617→
   618→   618→    if (hasUrl) memoryType = 'link'
   619→   619→    if (hasAudio && hasImage) memoryType = 'voice'
   620→   620→    if (!hasAudio && !hasImage && !hasUrl) memoryType = 'text'
   621→   621→
   622→   622→    // ── STEP 5: AI Cognitive Synthesis (z-ai-web-dev-sdk LLM) ──────
   623→   623→    // Determine initial title
   624→   624→    let aiTitle = hasText ? text.slice(0, 80) : hasUrl ? 'Saved Link' : hasAudio ? 'Voice Note' : 'Image Capture'
   625→   625→    let aiSummary: string | null = null
   626→   626→    let aiDeepInsight: string | null = null
   627→   627→    let aiTags = autoGenerateTags(rawContent, aiTitle) // keyword fallback always runs
   628→   628→    let connectedThemes: string[] = []
   629→   629→
   630→   630→    // Always run LLM synthesis (z-ai-web-dev-sdk is always available)
   631→   631→    const synthesis = await synthesizeWithLLM(rawContent)
   632→   632→    if (synthesis) {
   633→   633→      aiTitle = synthesis.suggested_title || aiTitle
   634→   634→      aiSummary = synthesis.summary
   635→   635→      aiDeepInsight = synthesis.deep_insight
   636→   636→      if (synthesis.tags.length > 0) {
   637→   637→        // Merge AI tags with keyword tags, deduplicate
   638→   638→        const allTags = [...new Set([...synthesis.tags, ...aiTags])]
   639→   639→        aiTags = allTags.slice(0, 6)
   640→   640→      }
   641→   641→      if (synthesis.connected_themes.length > 0) {
   642→   642→        connectedThemes = synthesis.connected_themes
   643→   643→      }
   644→   644→    }
   645→   645→
   646→   646→    // For images with VLM analysis, merge VLM tags too
   647→   647→    if (imageAnalysis && imageAnalysis.tags.length > 0) {
   648→   648→      const allTags = [...new Set([...aiTags, ...imageAnalysis.tags.map(t => t.toLowerCase())])]
   649→   649→      aiTags = allTags.slice(0, 6)
   650→   650→    }
   651→   651→
   652→   652→    // ── STEP 6: Find Connected Memories (AI Brain) ─────────────────
   653→   653→    let connectedMemories: { id: string; title: string; reason: string }[] = []
   654→   654→    try {
   655→   655→      connectedMemories = await findConnectedMemories(aiTags, rawContent, 'pending')
   656→   656→    } catch {
   657→   657→      // Non-critical — continue without connections
   658→   658→    }
   659→   659→
   660→   660→    // ── STEP 7: Try Supabase first (cookie-aware route handler client) ──
   661→   661→    if (isSupabaseConfigured()) {
   662→   662→      try {
   663→   663→        const supabase = await getSupabaseRouteClient()
   664→   664→        const { data: { user }, error: authError } = await supabase.auth.getUser()
   665→   665→
   666→   666→        if (!authError && user) {
   667→   667→          // Authenticated Supabase path
   668→   668→          // Upload image if present
   669→   669→          if (hasImage) {
   670→   670→            imageUrl = await uploadImageToStorage(imageFile, user.id, supabase)
   671→   671→          }
   672→   672→
   673→   673→          const { data: memoryRow, error: insertError } = await supabase
   674→   674→            .from('memories')
   675→   675→            .insert({
   676→   676→              user_id: user.id,
   677→   677→              type: memoryType,
   678→   678→              title: aiTitle,
   679→   679→              content: rawContent,
   680→   680→              summary: aiSummary,
   681→   681→              deep_insight: aiDeepInsight,
   682→   682→              tags: aiTags.join(','),
   683→   683→              source_url: hasUrl ? url.trim() : null,
   684→   684→              image_url: imageUrl,
   685→   685→              image_preview: null,
   686→   686→              file_url: null,
   687→   687→              recap: null,
   688→   688→              is_favorite: false,
   689→   689→            })
   690→   690→            .select('*, memory_collections(collection_id, collections(id, name, color, icon))')
   691→   691→            .single()
   692→   692→
   693→   693→          if (!insertError && memoryRow) {
   694→   694→            // Collection Tag Matching
   695→   695→            await matchOrCreateCollections(supabase, user.id, (memoryRow as { id: string }).id, aiTags)
   696→   696→
   697→   697→            // Return memory object in the format the frontend expects
   698→   698→            const row = memoryRow as Record<string, unknown>
   699→   699→            const memoryCollections = (row.memory_collections as Array<{ collection_id: string; collections: { id: string; name: string; color: string; icon: string } }>) || []
   700→   700→
   701→   701→            const memory = {
   702→   702→              id: row.id as string,
   703→   703→              type: (row.type as string) || 'text',
   704→   704→              title: (row.title as string) || '',
   705→   705→              content: (row.content as string) || '',
   706→   706→              summary: (row.summary as string) || null,
   707→   707→              deepInsight: (row.deep_insight as string) || null,
   708→   708→              tags: row.tags ? (row.tags as string).split(',').filter(Boolean) : [],
   709→   709→              sourceUrl: (row.source_url as string) || null,
   710→   710→              fileUrl: (row.file_url as string) || null,
   711→   711→              imagePreview: (row.image_preview as string) || null,
   712→   712→              imageUrl: (row.image_url as string) || null,
   713→   713→              recap: (row.recap as string) || null,
   714→   714→              isFavorite: (row.is_favorite as boolean) || false,
   715→   715→              createdAt: (row.created_at as string) || new Date().toISOString(),
   716→   716→              updatedAt: (row.updated_at as string) || new Date().toISOString(),
   717→   717→              collections: memoryCollections.map(mc => ({
   718→   718→                id: mc.collections.id,
   719→   719→                name: mc.collections.name,
   720→   720→                color: mc.collections.color,
   721→   721→                icon: mc.collections.icon,
   722→   722→              })),
   723→   723→              connectedMemories,
   724→   724→              connectedThemes,
   725→   725→            }
   726→   726→
   727→   727→            return NextResponse.json({ success: true, memory })
   728→   728→          }
   729→   729→
   730→   730→          // If Supabase insert failed, fall through to Prisma
   731→   731→          console.warn('Supabase insert failed, falling back to Prisma:', insertError?.message)
   732→   732→        }
   733→   733→      } catch (err) {
   734→   734→        // Supabase not working, fall through to Prisma
   735→   735→        console.warn('Supabase capture failed, falling back to Prisma:', err instanceof Error ? err.message : 'Unknown')
   736→   736→      }
   737→   737→    }
   738→   738→
   739→   739→    // ── STEP 8: Prisma Fallback (always works) ─────────────────────
   740→   740→    const memory = await saveToPrisma({
   741→   741→      type: memoryType,
   742→   742→      title: aiTitle,
   743→   743→      content: rawContent,
   744→   744→      summary: aiSummary,
   745→   745→      deepInsight: aiDeepInsight,
   746→   746→      tags: aiTags,
   747→   747→      sourceUrl: hasUrl ? url.trim() : null,
   748→   748→      imageUrl: imageUrl,
   749→   749→      imagePreview: null,
   750→   750→      recap: null,
   751→   751→    })
   752→   752→
   753→   753→    // Add connection data to response
   754→   754→    const memoryWithConnections = {
   755→   755→      ...memory,
   756→   756→      connectedMemories,
   757→   757→      connectedThemes,
   758→   758→    }
   759→   759→
   760→   760→    return NextResponse.json({ success: true, memory: memoryWithConnections })
   761→   761→  } catch (error) {
   762→   762→    console.error('Capture route error:', error)
   763→   763→    const message = error instanceof Error ? error.message : 'Internal server error'
   764→   764→    return NextResponse.json(
   765→   765→      { error: message },
   766→   766→      { status: 500 }
   767→   767→    )
   768→   768→  }
   769→   769→}
