   210→// ─── WEEKLY RECAP HELPER ─────────────────────────────────────────────
   211→// ═══════════════════════════════════════════════════════════════════════
   212→
   213→function getWeeklyRecap(memories: Memory[]): { count: number; topTags: string[]; summary: string } | null {
   214→  const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000
   215→  const weekMemories = memories.filter(m => new Date(m.createdAt).getTime() > sevenDaysAgo)
   216→  if (weekMemories.length === 0) return null
   217→
   218→  const tagCounts: Record<string, number> = {}
   219→  weekMemories.forEach(m => {
   220→    m.tags.forEach(t => {
   221→      tagCounts[t] = (tagCounts[t] || 0) + 1
   222→    })
   223→  })
   224→  const topTags = Object.entries(tagCounts)
   225→    .sort((a, b) => b[1] - a[1])
   226→    .slice(0, 3)
   227→    .map(([tag]) => tag)
   228→
   229→  const latestRecap = weekMemories[0]?.recap
   230→  if (latestRecap) {
   231→    return { count: weekMemories.length, topTags, summary: latestRecap.slice(0, 200) }
   232→  }
   233→
   234→  const summaries = weekMemories
   235→    .map(m => m.summary || m.content || m.title)
   236→    .filter(Boolean)
   237→    .slice(0, 5)
   238→  const combinedText = summaries.join('. ')
   239→  const summaryText = combinedText.length > 150
   240→    ? combinedText.slice(0, 147) + '...'
   241→    : combinedText || `You captured ${weekMemories.length} thought${weekMemories.length > 1 ? 's' : ''} this week.`
   242→
   243→  return { count: weekMemories.length, topTags, summary: summaryText }
   244→}
