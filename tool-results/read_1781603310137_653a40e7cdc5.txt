    22→  const key = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
    23→
    24→  const cookieStore = await cookies()

    27→    cookies: {
    28→      getAll() {
    29→        return cookieStore.getAll()
    30→      },
    31→      setAll(cookiesToSet) {
    32→        try {
    33→          cookiesToSet.forEach(({ name, value, options }) =>
    34→            cookieStore.set(name, value, options)
    35→          )
    36→        } catch {
    37→          // The `setAll` method was called from a Server Component.
    38→          // This can be ignored if you have middleware refreshing sessions.
    39→        }
    40→      },
    41→    },
    42→  })

   286→async function transcribeAudio(audioFile: File): Promise<string> {
   287→  // Try z-ai-web-dev-sdk ASR first (always available in this environment)
   288→  try {
   289→    const ZAI = (await import('z-ai-web-dev-sdk')).default
   290→    const sdk = await ZAI.create()

   292→    const base64Audio = Buffer.from(arrayBuffer).toString('base64')
   293→    const result = await sdk.audio.asr.create({
   294→      file_base64: base64Audio,
   295→    })
   296→    if (result && typeof result === 'object' && 'text' in result) {
   297→      const text = (result as { text: string }).text
   298→      if (text?.trim()) return text
   299→    }
   300→    if (typeof result === 'string' && result.trim()) return result
   301→  } catch (err) {
   302→    console.warn('z-ai-web-dev-sdk ASR failed:', err instanceof Error ? err.message : 'Unknown')
   303→  }
   304→
   305→  // Try Groq Whisper as fallback
   306→  const groqKey = process.env.NEXT_PUBLIC_GROQ_API_KEY
   307→  if (groqKey && groqKey !== 'placeholder_groq_key') {
   308→    try {
   309→      const formData = new FormData()
   310→      formData.append('file', audioFile)
   311→      formData.append('model', 'whisper-large-v3-turbo')
   312→      formData.append('response_format', 'json')
   313→
   314→      const response = await fetch('https://api.groq.com/openai/v1/audio/transcriptions', {
   315→        method: 'POST',
   316→        headers: {
   317→          'Authorization': `Bearer ${groqKey}`,
   318→        },
   319→        body: formData,
   320→      })
   321→
   322→      if (response.ok) {
   323→        const data = await response.json()
   324→        if (data.text?.trim()) return data.text
   325→      }
   326→    } catch (err) {
   327→      console.warn('Groq Whisper failed:', err instanceof Error ? err.message : 'Unknown')
   328→    }
   329→  }
   330→
   331→  return ''
   332→}

   481→// ═══════════════════════════════════════════════════════════════════════
   482→
   483→async function findConnectedMemories(
   484→  newMemoryTags: string[],
   485→  newMemoryContent: string,
   486→  newMemoryId: string
   487→): Promise<{ id: string; title: string; reason: string }[]> {
   488→  try {
   489→    // Get all existing memories from Prisma
   490→    const allMemories = await db.memory.findMany({
   491→      where: {
   492→        id: { not: newMemoryId },
   493→      },
   494→      select: {
   495→        id: true,
   496→        title: true,
   497→        content: true,
   498→        tags: true,
   499→        type: true,
   500→      },
   501→      take: 50,
   502→      orderBy: { createdAt: 'desc' },
   503→    })
   504→
   505→    if (allMemories.length === 0) return []
   506→
   507→    // ── Quick tag-based matching (instant, no LLM needed) ─────────
   508→    const connected: { id: string; title: string; reason: string; score: number }[] = []
   509→
   510→    for (const mem of allMemories) {
   511→      const memTags = mem.tags ? mem.tags.split(',').filter(Boolean) : []
   512→      const overlapTags = memTags.filter(t => newMemoryTags.includes(t.toLowerCase()))
   513→
   514→      if (overlapTags.length > 0) {
   515→        connected.push({
   516→          id: mem.id,
   517→          title: mem.title || 'Untitled',
   518→          reason: `Shares tags: ${overlapTags.join(', ')}`,
   519→          score: overlapTags.length * 2,
   520→        })
   521→      }
   522→
   523→      // Content similarity — check for common meaningful words
   524→      const contentWords = mem.content.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   525→      const newWords = newMemoryContent.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   526→      const overlapWords = contentWords.filter(w => newWords.includes(w))
   527→      if (overlapWords.length >= 2) {
   528→        const existing = connected.find(c => c.id === mem.id)
   529→        if (existing) {
   530→          existing.score += overlapWords.length
   531→        } else {
   532→          connected.push({
   533→            id: mem.id,
   534→            title: mem.title || 'Untitled',
   535→            reason: `Similar content themes`,
   536→            score: overlapWords.length,
   537→          })
   538→        }
   539→      }
   540→    }
   541→
   542→    // Sort by score and return top 5
   543→    return connected
   544→      .sort((a, b) => b.score - a.score)
   545→      .slice(0, 5)
   546→      .map(({ id, title, reason }) => ({ id, title, reason }))
   547→  } catch (err) {
   548→    console.warn('Find connected memories failed:', err instanceof Error ? err.message : 'Unknown')
   549→    return []
   550→  }
   551→}

   553→// ═══════════════════════════════════════════════════════════════════════
   554→// ─── BACKGROUND ENRICHMENT WORKER ────────────────────────────────────
   555→// ═══════════════════════════════════════════════════════════════════════
   556→// This function runs AFTER the response is sent. It performs all the
   557→// expensive AI operations (LLM synthesis, VLM analysis, transcription)
   558→// and then updates the database row with the enriched data.
   559→
   560→async function backgroundEnrichMemory(
   561→  memoryId: string,
   562→  rawContent: string,
   563→  memoryType: string,
   564→  imageFile: File | null,
   565→  audioFile: File | null,
   566→  hasUrl: boolean,
   567→  url: string,
   568→  supabaseUserId: string | null
   569→): Promise<void> {
   570→  try {
   571→    // ── 1. Audio transcription (if voice capture) ──────────────────
   572→    let audioTranscript = ''
   573→    if (audioFile && audioFile.size > 0) {
   574→      audioTranscript = await transcribeAudio(audioFile)
   575→    }
   576→
   577→    // ── 2. Image analysis (if image capture) ───────────────────────
   578→    let imageAnalysis: ImageAnalysis | null = null
   579→    if (imageFile && imageFile.size > 0) {
   580→      imageAnalysis = await analyzeImageWithVLM(imageFile)
   581→    }
   582→
   583→    // ── 3. Build enriched content ──────────────────────────────────
   584→    let enrichedContent = rawContent
   585→    if (audioTranscript) {
   586→      enrichedContent += (enrichedContent ? '\n\n' : '') + `Voice Transcript:\n${audioTranscript}`
   587→    }
   588→    if (imageAnalysis) {
   589→      const imageContentParts: string[] = []
   590→      if (imageAnalysis.description) imageContentParts.push(imageAnalysis.description)
   591→      if (imageAnalysis.extracted_text) imageContentParts.push(`Text in image: ${imageAnalysis.extracted_text}`)
   592→      if (imageAnalysis.objects.length > 0) imageContentParts.push(`Objects: ${imageAnalysis.objects.join(', ')}`)
   593→      if (imageContentParts.length > 0) {
   594→        enrichedContent += (enrichedContent ? '\n\n' : '') + imageContentParts.join('\n\n')
   595→      }
   596→    }
   597→
   598→    // ── 4. AI Cognitive Synthesis ──────────────────────────────────
   599→    const synthesis = await synthesizeWithLLM(enrichedContent)
   600→
   601→    let aiTitle: string | null = null
   602→    let aiSummary: string | null = null
   603→    let aiDeepInsight: string | null = null
   604→    let aiTags = autoGenerateTags(enrichedContent, rawContent.slice(0, 80))
   605→    let connectedThemes: string[] = []
   606→
   607→    if (synthesis) {
   608→      aiTitle = synthesis.suggested_title
   609→      aiSummary = synthesis.summary
   610→      aiDeepInsight = synthesis.deep_insight
   611→      if (synthesis.tags.length > 0) {
   612→        const allTags = [...new Set([...synthesis.tags, ...aiTags])]
   613→        aiTags = allTags.slice(0, 6)
   614→      }
   615→      if (synthesis.connected_themes.length > 0) {
   616→        connectedThemes = synthesis.connected_themes
   617→      }
   618→    }
   619→
   620→    // Merge VLM tags
   621→    if (imageAnalysis && imageAnalysis.tags.length > 0) {
   622→      const allTags = [...new Set([...aiTags, ...imageAnalysis.tags.map(t => t.toLowerCase())])]
   623→      aiTags = allTags.slice(0, 6)
   624→    }
   625→
   626→    // ── 5. Upload image to Supabase Storage (if applicable) ────────
   627→    let uploadedImageUrl: string | null = null
   628→    if (imageFile && imageFile.size > 0 && supabaseUserId && isSupabaseConfigured()) {
   629→      try {
   630→        const supabase = await getSupabaseRouteClient()
   631→        uploadedImageUrl = await uploadImageToStorage(imageFile, supabaseUserId, supabase)
   632→      } catch (err) {
   633→        console.warn('Background image upload failed:', err instanceof Error ? err.message : 'Unknown')
   634→      }
   635→    }
   636→
   637→    // ── 6. Update database with enriched data ─────────────────────
   638→    const updateData: Record<string, unknown> = {
   639→      content: enrichedContent,
   640→      tags: aiTags.join(','),
   641→    }
   642→    if (aiTitle) updateData.title = aiTitle
   643→    if (aiSummary) updateData.summary = aiSummary
   644→    if (aiDeepInsight) updateData.deep_insight = aiDeepInsight
   645→    if (uploadedImageUrl) updateData.image_url = uploadedImageUrl
   646→
   647→    // Try Supabase update first
   648→    if (supabaseUserId && isSupabaseConfigured()) {
   649→      try {
   650→        const supabase = await getSupabaseRouteClient()
   651→        const { error } = await supabase
   652→          .from('memories')
   653→          .update(updateData)
   654→          .eq('id', memoryId)
   655→
   656→        if (!error) {
   657→          // Collection tag matching
   658→          await matchOrCreateCollections(supabase, supabaseUserId, memoryId, aiTags)
   659→
   660→          // Trigger embedding generation in background
   661→          try {
   662→            const embeddingContent = [aiTitle, aiSummary, enrichedContent].filter(Boolean).join(' ').slice(0, 2000)
   663→            await fetch('/api/generate-embedding', {
   664→              method: 'POST',
   665→              headers: { 'Content-Type': 'application/json' },
   666→              body: JSON.stringify({ memoryId, content: embeddingContent }),
   667→            })
   668→          } catch {
   669→            // Non-blocking — embedding is nice-to-have
   670→          }
   671→
   672→          console.log(`[Background Enrichment] Memory ${memoryId} enriched successfully (Supabase)`)
   673→          return
   674→        }
   675→        console.warn('Supabase enrichment update failed:', error.message)
   676→      } catch (err) {
   677→        console.warn('Supabase enrichment failed, falling back to Prisma:', err instanceof Error ? err.message : 'Unknown')
   678→      }
   679→    }
   680→
   681→    // Prisma fallback for enrichment update
   682→    try {
   683→      const prismaUpdateData: Record<string, unknown> = {
   684→        content: enrichedContent,
   685→        tags: aiTags.join(','),
   686→      }
   687→      if (aiTitle) prismaUpdateData.title = aiTitle
   688→      if (aiSummary) prismaUpdateData.summary = aiSummary
   689→      if (aiDeepInsight) prismaUpdateData.deepInsight = aiDeepInsight
   690→      if (uploadedImageUrl) prismaUpdateData.imageUrl = uploadedImageUrl
   691→
   692→      await db.memory.update({
   693→        where: { id: memoryId },
   694→        data: prismaUpdateData,
   695→      })
   696→
   697→      console.log(`[Background Enrichment] Memory ${memoryId} enriched successfully (Prisma)`)
   698→    } catch (err) {
   699→      console.warn('Prisma enrichment update failed:', err instanceof Error ? err.message : 'Unknown')
   700→    }
   701→  } catch (err) {
   702→    console.error('[Background Enrichment] Fatal error:', err instanceof Error ? err.message : 'Unknown')
   703→  }
   704→}

   706→// ═══════════════════════════════════════════════════════════════════════
   707→// ─── MAIN POST HANDLER ──────────────────────────────────────────────
   708→// ═══════════════════════════════════════════════════════════════════════
   709→//
   710→// ARCHITECTURE: Immediate Ingestion + Asynchronous Enrichment
   711→//
   712→// 1. Parse FormData → validate → save raw row to DB → return instantly (~200ms)
   713→// 2. After response: fire background AI (LLM, VLM, ASR) → update row with enriched data
   714→//
   715→// The frontend uses Optimistic UI to show a placeholder immediately,
   716→// then replaces it with the real memory when the API responds, and
   717→// later updates it again when AI enrichment completes via polling or refetch.
   718→
   719→export async function POST(req: NextRequest) {
   720→  try {
   721→    // ── STEP 1: Parse FormData ──────────────────────────────────────
   722→    const formData = await req.formData()
   723→    const text = (formData.get('text') as string) || ''
   724→    const url = (formData.get('url') as string) || ''
   725→    const imageFile = formData.get('image') as File | null
   726→    const audioFile = formData.get('audio') as File | null
   727→
   728→    // Validate: at least one content source
   729→    const hasText = text.trim().length > 0
   730→    const hasUrl = url.trim().length > 0
   731→    const hasImage = imageFile && imageFile.size > 0
   732→    const hasAudio = audioFile && audioFile.size > 0
   733→
   734→    if (!hasText && !hasUrl && !hasImage && !hasAudio) {
   735→      return NextResponse.json({ error: 'No content provided' }, { status: 400 })
   736→    }
   737→
   738→    // ── STEP 2: Determine type & build raw content ─────────────────
   739→    let memoryType: string = 'text'
   740→    if (hasAudio) memoryType = 'voice'
   741→    if (hasImage) memoryType = 'image'
   742→    if (hasUrl) memoryType = 'link'
   743→    if (hasAudio && hasImage) memoryType = 'voice'
   744→    if (!hasAudio && !hasImage && !hasUrl) memoryType = 'text'
   745→
   746→    // Build raw content from what we have immediately
   747→    let rawContent = ''
   748→    if (hasText) rawContent += text.trim()
   749→    if (hasUrl) rawContent += (rawContent ? '\n\n' : '') + `URL: ${url.trim()}`
   750→    if (hasImage && !rawContent) rawContent = 'Image capture'
   751→    if (hasAudio && !rawContent) rawContent = 'Voice note'
   752→    if (!rawContent) rawContent = 'Captured content'
   753→
   754→    // ── STEP 3: Instant keyword-based tags (no AI, zero latency) ───
   755→    const instantTags = autoGenerateTags(rawContent, text.slice(0, 80))
   756→
   757→    // ── STEP 4: Immediate DB save (raw, no AI enrichment yet) ──────
   758→    let savedMemoryId: string
   759→    let supabaseUserId: string | null = null
   760→    let savedMemory: Record<string, unknown>
   761→
   762→    // Try Supabase first
   763→    if (isSupabaseConfigured()) {
   764→      try {
   765→        const supabase = await getSupabaseRouteClient()
   766→        const { data: { user }, error: authError } = await supabase.auth.getUser()
   767→
   768→        if (!authError && user) {
   769→          supabaseUserId = user.id
   770→
   771→          // Upload image immediately (storage is fast, doesn't need AI)
   772→          let imageUrl: string | null = null
   773→          if (hasImage) {
   774→            imageUrl = await uploadImageToStorage(imageFile, user.id, supabase)
   775→          }
   776→
   777→          const { data: memoryRow, error: insertError } = await supabase
   778→            .from('memories')
   779→            .insert({
   780→              user_id: user.id,
   781→              type: memoryType,
   782→              title: hasText ? text.slice(0, 80) : hasUrl ? 'Saved Link' : hasAudio ? 'Voice Note' : 'Image Capture',
   783→              content: rawContent,
   784→              summary: null, // Will be filled by background enrichment
   785→              deep_insight: null,
   786→              tags: instantTags.join(','),
   787→              source_url: hasUrl ? url.trim() : null,
   788→              image_url: imageUrl,
   789→              image_preview: null,
   790→              file_url: null,
   791→              recap: null,
   792→              is_favorite: false,
   793→            })
   794→            .select('*, memory_collections(collection_id, collections(id, name, color, icon))')
   795→            .single()
   796→
   797→          if (!insertError && memoryRow) {
   798→            const row = memoryRow as Record<string, unknown>
   799→            savedMemoryId = row.id as string
   800→            const memoryCollections = (row.memory_collections as Array<{ collection_id: string; collections: { id: string; name: string; color: string; icon: string } }>) || []
   801→
   802→            savedMemory = {
   803→              id: row.id,
   804→              type: (row.type as string) || 'text',
   805→              title: (row.title as string) || '',
   806→              content: (row.content as string) || '',
   807→              summary: (row.summary as string) || null,
   808→              deepInsight: (row.deep_insight as string) || null,
   809→              tags: row.tags ? (row.tags as string).split(',').filter(Boolean) : [],
   810→              sourceUrl: (row.source_url as string) || null,
   811→              fileUrl: (row.file_url as string) || null,
   812→              imagePreview: (row.image_preview as string) || null,
   813→              imageUrl: (row.image_url as string) || null,
   814→              recap: (row.recap as string) || null,
   815→              isFavorite: (row.is_favorite as boolean) || false,
   816→              createdAt: (row.created_at as string) || new Date().toISOString(),
   817→              updatedAt: (row.updated_at as string) || new Date().toISOString(),
   818→              collections: memoryCollections.map(mc => ({
   819→                id: mc.collections.id,
   820→                name: mc.collections.name,
   821→                color: mc.collections.color,
   822→                icon: mc.collections.icon,
   823→              })),
   824→              enriching: true, // Signal to frontend that AI enrichment is pending
   825→            }
   826→
   827→            // ── INSTANT RETURN — user sees memory immediately ────────
   828→            const response = NextResponse.json({ success: true, memory: savedMemory })
   829→
   830→            // ── BACKGROUND ENRICHMENT (fire and forget) ──────────────
   831→            // This runs AFTER the response is sent to the client
   832→            backgroundEnrichMemory(
   833→              savedMemoryId,
   834→              rawContent,
   835→              memoryType,
   836→              hasImage ? imageFile : null,
   837→              hasAudio ? audioFile : null,
   838→              hasUrl,
   839→              url,
   840→              supabaseUserId
   841→            ).catch(err => {
   842→              console.error('[Background Enrichment] Unhandled error:', err instanceof Error ? err.message : 'Unknown')
   843→            })
   844→
   845→            return response
   846→          }
   847→
   848→          console.warn('Supabase insert failed, falling back to Prisma:', insertError?.message)
   849→        }
   850→      } catch (err) {
   851→        console.warn('Supabase capture failed, falling back to Prisma:', err instanceof Error ? err.message : 'Unknown')
   852→      }
   853→    }
   854→
   855→    // ── PRISMA FALLBACK (always works) ─────────────────────────────
   856→    const memory = await saveToPrisma({
   857→      type: memoryType,
   858→      title: hasText ? text.slice(0, 80) : hasUrl ? 'Saved Link' : hasAudio ? 'Voice Note' : 'Image Capture',
   859→      content: rawContent,
   860→      summary: null, // Will be filled by background enrichment
   861→      deepInsight: null,
   862→      tags: instantTags,
   863→      sourceUrl: hasUrl ? url.trim() : null,
   864→      imageUrl: null,
   865→      imagePreview: null,
   866→      recap: null,
   867→    })
   868→
   869→    savedMemoryId = memory.id
   870→    savedMemory = {
   871→      ...memory,
   872→      enriching: true, // Signal to frontend that AI enrichment is pending
   873→    }
   874→
   875→    // ── INSTANT RETURN — user sees memory immediately ────────────────
   876→    const response = NextResponse.json({ success: true, memory: savedMemory })
   877→
   878→    // ── BACKGROUND ENRICHMENT (fire and forget) ──────────────────────
   879→    backgroundEnrichMemory(
   880→      savedMemoryId,
   881→      rawContent,
   882→      memoryType,
   883→      hasImage ? imageFile : null,
   884→      hasAudio ? audioFile : null,
   885→      hasUrl,
   886→      url,
   887→      supabaseUserId
   888→    ).catch(err => {
   889→      console.error('[Background Enrichment] Unhandled error:', err instanceof Error ? err.message : 'Unknown')
   890→    })
   891→
   892→    return response
   893→  } catch (error) {
   894→    console.error('Capture route error:', error)
   895→    const message = error instanceof Error ? error.message : 'Internal server error'
   896→    return NextResponse.json(
   897→      { error: message },
   898→      { status: 500 }
   899→    )
   900→  }
   901→}
