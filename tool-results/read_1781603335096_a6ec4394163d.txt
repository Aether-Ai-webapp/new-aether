   402→   402→              memory_id: memoryId,
   403→   403→              collection_id: matched.id,
   404→   404→            })

   406→   406→        }
   407→   407→      }
   408→   408→    }

   481→   481→// ═══════════════════════════════════════════════════════════════════════
   482→   482→
   483→   483→async function findConnectedMemories(
   484→   484→  newMemoryTags: string[],
   485→   485→  newMemoryContent: string,
   486→   486→  newMemoryId: string
   487→   487→): Promise<{ id: string; title: string; reason: string }[]> {
   488→   488→  try {
   489→   489→    // Get all existing memories from Prisma
   490→   490→    const allMemories = await db.memory.findMany({
   491→   491→      where: {
   492→   492→        id: { not: newMemoryId },
   493→   493→      },
   494→   494→      select: {
   495→   495→        id: true,
   496→   496→        title: true,
   497→   497→        content: true,
   498→   498→        tags: true,
   499→   499→        type: true,
   500→   500→      },
   501→   501→      take: 50,
   502→   502→      orderBy: { createdAt: 'desc' },
   503→   503→    })
   504→   504→
   505→   505→    if (allMemories.length === 0) return []
   506→   506→
   507→   507→    // ── Quick tag-based matching (instant, no LLM needed) ─────────
   508→   508→    const connected: { id: string; title: string; reason: string; score: number }[] = []
   509→   509→
   510→   510→    for (const mem of allMemories) {
   511→   511→      const memTags = mem.tags ? mem.tags.split(',').filter(Boolean) : []
   512→   512→      const overlapTags = memTags.filter(t => newMemoryTags.includes(t.toLowerCase()))
   513→   513→
   514→   514→      if (overlapTags.length > 0) {
   515→   515→        connected.push({
   516→   516→          id: mem.id,
   517→   517→          title: mem.title || 'Untitled',
   518→   518→          reason: `Shares tags: ${overlapTags.join(', ')}`,
   519→   519→          score: overlapTags.length * 2,
   520→   520→        })
   521→   521→      }
   522→   522→
   523→   523→      // Content similarity — check for common meaningful words
   524→   524→      const contentWords = mem.content.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   525→   525→      const newWords = newMemoryContent.toLowerCase().split(/\s+/).filter(w => w.length > 4)
   526→   526→      const overlapWords = contentWords.filter(w => newWords.includes(w))
   527→   527→      if (overlapWords.length >= 2) {
   528→   528→        const existing = connected.find(c => c.id === mem.id)
   529→   529→        if (existing) {
   530→   530→          existing.score += overlapWords.length
   531→   531→        } else {
   532→   532→          connected.push({
   533→   533→            id: mem.id,
   534→   534→            title: mem.title || 'Untitled',
   535→   535→            reason: `Similar content themes`,
   536→   536→            score: overlapWords.length,
   537→   537→          })
   538→   538→        }
   539→   539→      }
   540→   540→    }
   541→   541→
   542→   542→    // Sort by score and return top 5
   543→   543→    return connected
   544→   544→      .sort((a, b) => b.score - a.score)
   545→   545→      .slice(0, 5)
   546→   546→      .map(({ id, title, reason }) => ({ id, title, reason }))
   547→   547→  } catch (err) {
   548→   548→    console.warn('Find connected memories failed:', err instanceof Error ? err.message : 'Unknown')
   549→   549→    return []
   550→   550→  }
   551→   551→}

   553→   553→// ═══════════════════════════════════════════════════════════════════════
   554→   554→// ─── BACKGROUND ENRICHMENT WORKER ────────────────────────────────────
   555→   555→// ═══════════════════════════════════════════════════════════════════════
   556→   556→// This function runs AFTER the response is sent. It performs all the
   557→   557→// expensive AI operations (LLM synthesis, VLM analysis, transcription)
   558→   558→// and then updates the database row with the enriched data.
   559→   559→
   560→   560→async function backgroundEnrichMemory(
   561→   561→  memoryId: string,
   562→   562→  rawContent: string,
   563→   563→  memoryType: string,
   564→   564→  imageFile: File | null,
   565→   565→  audioFile: File | null,
   566→   566→  hasUrl: boolean,
   567→   567→  url: string,
   568→   568→  supabaseUserId: string | null
   569→   569→): Promise<void> {
   570→   570→  try {
   571→   571→    // ── 1. Audio transcription (if voice capture) ──────────────────
   572→   572→    let audioTranscript = ''
   573→   573→    if (audioFile && audioFile.size > 0) {
   574→   574→      audioTranscript = await transcribeAudio(audioFile)
   575→   575→    }
   576→   576→
   577→   577→    // ── 2. Image analysis (if image capture) ───────────────────────
   578→   578→    let imageAnalysis: ImageAnalysis | null = null
   579→   579→    if (imageFile && imageFile.size > 0) {
   580→   580→      imageAnalysis = await analyzeImageWithVLM(imageFile)
   581→   581→    }
   582→   582→
   583→   583→    // ── 3. Build enriched content ──────────────────────────────────
   584→   584→    let enrichedContent = rawContent
   585→   585→    if (audioTranscript) {
   586→   586→      enrichedContent += (enrichedContent ? '\n\n' : '') + `Voice Transcript:\n${audioTranscript}`
   587→   587→    }
   588→   588→    if (imageAnalysis) {
   589→   589→      const imageContentParts: string[] = []
   590→   590→      if (imageAnalysis.description) imageContentParts.push(imageAnalysis.description)
   591→   591→      if (imageAnalysis.extracted_text) imageContentParts.push(`Text in image: ${imageAnalysis.extracted_text}`)
   592→   592→      if (imageAnalysis.objects.length > 0) imageContentParts.push(`Objects: ${imageAnalysis.objects.join(', ')}`)
   593→   593→      if (imageContentParts.length > 0) {
   594→   594→        enrichedContent += (enrichedContent ? '\n\n' : '') + imageContentParts.join('\n\n')
   595→   595→      }
   596→   596→    }
   597→   597→
   598→   598→    // ── 4. AI Cognitive Synthesis ──────────────────────────────────
   599→   599→    const synthesis = await synthesizeWithLLM(enrichedContent)
   600→   600→
   601→   601→    let aiTitle: string | null = null
   602→   602→    let aiSummary: string | null = null
   603→   603→    let aiDeepInsight: string | null = null
   604→   604→    let aiTags = autoGenerateTags(enrichedContent, rawContent.slice(0, 80))
   605→   605→    let connectedThemes: string[] = []
   606→   606→
   607→   607→    if (synthesis) {
   608→   608→      aiTitle = synthesis.suggested_title
   609→   609→      aiSummary = synthesis.summary
   610→   610→      aiDeepInsight = synthesis.deep_insight
   611→   611→      if (synthesis.tags.length > 0) {
   612→   612→        const allTags = [...new Set([...synthesis.tags, ...aiTags])]
   613→   613→        aiTags = allTags.slice(0, 6)
   614→   614→      }
   615→   615→      if (synthesis.connected_themes.length > 0) {
   616→   616→        connectedThemes = synthesis.connected_themes
   617→   617→      }
   618→   618→    }
   619→   619→
   620→   620→    // Merge VLM tags
   621→   621→    if (imageAnalysis && imageAnalysis.tags.length > 0) {
   622→   622→      const allTags = [...new Set([...aiTags, ...imageAnalysis.tags.map(t => t.toLowerCase())])]
   623→   623→      aiTags = allTags.slice(0, 6)
   624→   624→    }
   625→   625→
   626→   626→    // ── 5. Upload image to Supabase Storage (if applicable) ────────
   627→   627→    let uploadedImageUrl: string | null = null
   628→   628→    if (imageFile && imageFile.size > 0 && supabaseUserId && isSupabaseConfigured()) {
   629→   629→      try {
   630→   630→        const supabase = await getSupabaseRouteClient()
   631→   631→        uploadedImageUrl = await uploadImageToStorage(imageFile, supabaseUserId, supabase)
   632→   632→      } catch (err) {
   633→   633→        console.warn('Background image upload failed:', err instanceof Error ? err.message : 'Unknown')
   634→   634→      }
   635→   635→    }
   636→   636→
   637→   637→    // ── 6. Update database with enriched data ─────────────────────
   638→   638→    const updateData: Record<string, unknown> = {
   639→   639→      content: enrichedContent,
   640→   640→      tags: aiTags.join(','),
   641→   641→    }
   642→   642→    if (aiTitle) updateData.title = aiTitle
   643→   643→    if (aiSummary) updateData.summary = aiSummary
   644→   644→    if (aiDeepInsight) updateData.deep_insight = aiDeepInsight
   645→   645→    if (uploadedImageUrl) updateData.image_url = uploadedImageUrl
   646→   646→
   647→   647→    // Try Supabase update first
   648→   648→    if (supabaseUserId && isSupabaseConfigured()) {
   649→   649→      try {
   650→   650→        const supabase = await getSupabaseRouteClient()
   651→   651→        const { error } = await supabase
   652→   652→          .from('memories')
   653→   653→          .update(updateData)
   654→   654→          .eq('id', memoryId)
   655→   655→
   656→   656→        if (!error) {
   657→   657→          // Collection tag matching
   658→   658→          await matchOrCreateCollections(supabase, supabaseUserId, memoryId, aiTags)
   659→   659→
   660→   660→          // Trigger embedding generation in background
   661→   661→          try {
   662→   662→            const embeddingContent = [aiTitle, aiSummary, enrichedContent].filter(Boolean).join(' ').slice(0, 2000)
   663→   663→            await fetch('/api/generate-embedding', {
   664→   664→              method: 'POST',
   665→   665→              headers: { 'Content-Type': 'application/json' },
   666→   666→              body: JSON.stringify({ memoryId, content: embeddingContent }),
   667→   667→            })
   668→   668→          } catch {
   669→   669→            // Non-blocking — embedding is nice-to-have
   670→   670→          }
   671→   671→
   672→   672→          console.log(`[Background Enrichment] Memory ${memoryId} enriched successfully (Supabase)`)
   673→   673→          return
   674→   674→        }
   675→   675→        console.warn('Supabase enrichment update failed:', error.message)
   676→   676→      } catch (err) {
   677→   677→        console.warn('Supabase enrichment failed, falling back to Prisma:', err instanceof Error ? err.message : 'Unknown')
   678→   678→      }
   679→   679→    }
   680→   680→
   681→   681→    // Prisma fallback for enrichment update
   682→   682→    try {
   683→   683→      const prismaUpdateData: Record<string, unknown> = {
   684→   684→        content: enrichedContent,
   685→   685→        tags: aiTags.join(','),
   686→   686→      }
   687→   687→      if (aiTitle) prismaUpdateData.title = aiTitle
   688→   688→      if (aiSummary) prismaUpdateData.summary = aiSummary
   689→   689→      if (aiDeepInsight) prismaUpdateData.deepInsight = aiDeepInsight
   690→   690→      if (uploadedImageUrl) prismaUpdateData.imageUrl = uploadedImageUrl
   691→   691→
   692→   692→      await db.memory.update({
   693→   693→        where: { id: memoryId },
   694→   694→        data: prismaUpdateData,
   695→   695→      })
   696→   696→
   697→   697→      console.log(`[Background Enrichment] Memory ${memoryId} enriched successfully (Prisma)`)
   698→   698→    } catch (err) {
   699→   699→      console.warn('Prisma enrichment update failed:', err instanceof Error ? err.message : 'Unknown')
   700→   700→    }
   701→   701→  } catch (err) {
   702→   702→    console.error('[Background Enrichment] Fatal error:', err instanceof Error ? err.message : 'Unknown')
   703→   703→  }
   704→   704→}

   706→   706→// ═══════════════════════════════════════════════════════════════════════
   707→   707→// ─── MAIN POST HANDLER ──────────────────────────────────────────────
   708→   708→// ═══════════════════════════════════════════════════════════════════════
   709→   709→//
   710→   710→// ARCHITECTURE: Immediate Ingestion + Asynchronous Enrichment
   711→   711→//
   712→   712→// 1. Parse FormData → validate → save raw row to DB → return instantly (~200ms)
   713→   713→// 2. After response: fire background AI (LLM, VLM, ASR) → update row with enriched data
   714→   714→//
   715→   715→// The frontend uses Optimistic UI to show a placeholder immediately,
   716→   716→// then replaces it with the real memory when the API responds, and
   717→   717→// later updates it again when AI enrichment completes via polling or refetch.
   718→   718→
   719→   719→export async function POST(req: NextRequest) {
   720→   720→  try {
   721→   721→    // ── STEP 1: Parse FormData ──────────────────────────────────────
   722→   722→    const formData = await req.formData()
   723→   723→    const text = (formData.get('text') as string) || ''
   724→   724→    const url = (formData.get('url') as string) || ''
   725→   725→    const imageFile = formData.get('image') as File | null
   726→   726→    const audioFile = formData.get('audio') as File | null
   727→   727→
   728→   728→    // Validate: at least one content source
   729→   729→    const hasText = text.trim().length > 0
   730→   730→    const hasUrl = url.trim().length > 0
   731→   731→    const hasImage = imageFile && imageFile.size > 0
   732→   732→    const hasAudio = audioFile && audioFile.size > 0
   733→   733→
   734→   734→    if (!hasText && !hasUrl && !hasImage && !hasAudio) {
   735→   735→      return NextResponse.json({ error: 'No content provided' }, { status: 400 })
   736→   736→    }
   737→   737→
   738→   738→    // ── STEP 2: Determine type & build raw content ─────────────────
   739→   739→    let memoryType: string = 'text'
   740→   740→    if (hasAudio) memoryType = 'voice'
   741→   741→    if (hasImage) memoryType = 'image'
   742→   742→    if (hasUrl) memoryType = 'link'
   743→   743→    if (hasAudio && hasImage) memoryType = 'voice'
   744→   744→    if (!hasAudio && !hasImage && !hasUrl) memoryType = 'text'
   745→   745→
   746→   746→    // Build raw content from what we have immediately
   747→   747→    let rawContent = ''
   748→   748→    if (hasText) rawContent += text.trim()
   749→   749→    if (hasUrl) rawContent += (rawContent ? '\n\n' : '') + `URL: ${url.trim()}`
   750→   750→    if (hasImage && !rawContent) rawContent = 'Image capture'
   751→   751→    if (hasAudio && !rawContent) rawContent = 'Voice note'
   752→   752→    if (!rawContent) rawContent = 'Captured content'
   753→   753→
   754→   754→    // ── STEP 3: Instant keyword-based tags (no AI, zero latency) ───
   755→   755→    const instantTags = autoGenerateTags(rawContent, text.slice(0, 80))
   756→   756→
   757→   757→    // ── STEP 4: Immediate DB save (raw, no AI enrichment yet) ──────
   758→   758→    let savedMemoryId: string
   759→   759→    let supabaseUserId: string | null = null
   760→   760→    let savedMemory: Record<string, unknown>
   761→   761→
   762→   762→    // Try Supabase first
   763→   763→    if (isSupabaseConfigured()) {
   764→   764→      try {
   765→   765→        const supabase = await getSupabaseRouteClient()
   766→   766→        const { data: { user }, error: authError } = await supabase.auth.getUser()
   767→   767→
   768→   768→        if (!authError && user) {
   769→   769→          supabaseUserId = user.id
   770→   770→
   771→   771→          // Upload image immediately (storage is fast, doesn't need AI)
   772→   772→          let imageUrl: string | null = null
   773→   773→          if (hasImage) {
   774→   774→            imageUrl = await uploadImageToStorage(imageFile, user.id, supabase)
   775→   775→          }
   776→   776→
   777→   777→          const { data: memoryRow, error: insertError } = await supabase
   778→   778→            .from('memories')
   779→   779→            .insert({
   780→   780→              user_id: user.id,
   781→   781→              type: memoryType,
   782→   782→              title: hasText ? text.slice(0, 80) : hasUrl ? 'Saved Link' : hasAudio ? 'Voice Note' : 'Image Capture',
   783→   783→              content: rawContent,
   784→   784→              summary: null, // Will be filled by background enrichment
   785→   785→              deep_insight: null,
   786→   786→              tags: instantTags.join(','),
   787→   787→              source_url: hasUrl ? url.trim() : null,
   788→   788→              image_url: imageUrl,
   789→   789→              image_preview: null,
   790→   790→              file_url: null,
   791→   791→              recap: null,
   792→   792→              is_favorite: false,
   793→   793→            })
   794→   794→            .select('*, memory_collections(collection_id, collections(id, name, color, icon))')
   795→   795→            .single()
   796→   796→
   797→   797→          if (!insertError && memoryRow) {
   798→   798→            const row = memoryRow as Record<string, unknown>
   799→   799→            savedMemoryId = row.id as string
   800→   800→            const memoryCollections = (row.memory_collections as Array<{ collection_id: string; collections: { id: string; name: string; color: string; icon: string } }>) || []
   801→   801→
   802→   802→            savedMemory = {
   803→   803→              id: row.id,
   804→   804→              type: (row.type as string) || 'text',
   805→   805→              title: (row.title as string) || '',
   806→   806→              content: (row.content as string) || '',
   807→   807→              summary: (row.summary as string) || null,
   808→   808→              deepInsight: (row.deep_insight as string) || null,
   809→   809→              tags: row.tags ? (row.tags as string).split(',').filter(Boolean) : [],
   810→   810→              sourceUrl: (row.source_url as string) || null,
   811→   811→              fileUrl: (row.file_url as string) || null,
   812→   812→              imagePreview: (row.image_preview as string) || null,
   813→   813→              imageUrl: (row.image_url as string) || null,
   814→   814→              recap: (row.recap as string) || null,
   815→   815→              isFavorite: (row.is_favorite as boolean) || false,
   816→   816→              createdAt: (row.created_at as string) || new Date().toISOString(),
   817→   817→              updatedAt: (row.updated_at as string) || new Date().toISOString(),
   818→   818→              collections: memoryCollections.map(mc => ({
   819→   819→                id: mc.collections.id,
   820→   820→                name: mc.collections.name,
   821→   821→                color: mc.collections.color,
   822→   822→                icon: mc.collections.icon,
   823→   823→              })),
   824→   824→              enriching: true, // Signal to frontend that AI enrichment is pending
   825→   825→            }
   826→   826→
   827→   827→            // ── INSTANT RETURN — user sees memory immediately ────────
   828→   828→            const response = NextResponse.json({ success: true, memory: savedMemory })
   829→   829→
   830→   830→            // ── BACKGROUND ENRICHMENT (fire and forget) ──────────────
   831→   831→            // This runs AFTER the response is sent to the client
   832→   832→            backgroundEnrichMemory(
   833→   833→              savedMemoryId,
   834→   834→              rawContent,
   835→   835→              memoryType,
   836→   836→              hasImage ? imageFile : null,
   837→   837→              hasAudio ? audioFile : null,
   838→   838→              hasUrl,
   839→   839→              url,
   840→   840→              supabaseUserId
   841→   841→            ).catch(err => {
   842→   842→              console.error('[Background Enrichment] Unhandled error:', err instanceof Error ? err.message : 'Unknown')
   843→   843→            })
   844→   844→
   845→   845→            return response
   846→   846→          }
   847→   847→
   848→   848→          console.warn('Supabase insert failed, falling back to Prisma:', insertError?.message)
   849→   849→        }
   850→   850→      } catch (err) {
   851→   851→        console.warn('Supabase capture failed, falling back to Prisma:', err instanceof Error ? err.message : 'Unknown')
   852→   852→      }
   853→   853→    }
   854→   854→
   855→   855→    // ── PRISMA FALLBACK (always works) ─────────────────────────────
   856→   856→    const memory = await saveToPrisma({
   857→   857→      type: memoryType,
   858→   858→      title: hasText ? text.slice(0, 80) : hasUrl ? 'Saved Link' : hasAudio ? 'Voice Note' : 'Image Capture',
   859→   859→      content: rawContent,
   860→   860→      summary: null, // Will be filled by background enrichment
   861→   861→      deepInsight: null,
   862→   862→      tags: instantTags,
   863→   863→      sourceUrl: hasUrl ? url.trim() : null,
   864→   864→      imageUrl: null,
   865→   865→      imagePreview: null,
   866→   866→      recap: null,
   867→   867→    })
   868→   868→
   869→   869→    savedMemoryId = memory.id
   870→   870→    savedMemory = {
   871→   871→      ...memory,
   872→   872→      enriching: true, // Signal to frontend that AI enrichment is pending
   873→   873→    }
   874→   874→
   875→   875→    // ── INSTANT RETURN — user sees memory immediately ────────────────
   876→   876→    const response = NextResponse.json({ success: true, memory: savedMemory })
   877→   877→
   878→   878→    // ── BACKGROUND ENRICHMENT (fire and forget) ──────────────────────
   879→   879→    backgroundEnrichMemory(
   880→   880→      savedMemoryId,
   881→   881→      rawContent,
   882→   882→      memoryType,
   883→   883→      hasImage ? imageFile : null,
   884→   884→      hasAudio ? audioFile : null,
   885→   885→      hasUrl,
   886→   886→      url,
   887→   887→      supabaseUserId
   888→   888→    ).catch(err => {
   889→   889→      console.error('[Background Enrichment] Unhandled error:', err instanceof Error ? err.message : 'Unknown')
